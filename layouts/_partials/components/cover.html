{{/* 程序化排版封面 ---------------- 零素材依赖：按主题 + 日期派生种子，从 5 色调色板取沉稳底色， 以「超大衬线首字 + 刊名小字 + 内衬细线」构成排版封面。 3 种字位布局提供变化，全部依赖排版而非几何图形。 遵循明暗主题（调色板变量随 dark mode 切换）。 用法：{{ partial "components/cover" . }} 可选：{{ partial "components/cover" (dict "ctx" . "ratio" "16:9") }} */}} {{- $ctx := .ctx -}} {{- $ratio := .ratio | default "16:9" -}} {{- $title := printf "%v" $ctx.Title -}} {{- $key := $ctx.Section | default "misc" -}} {{/* categories / tags 可能是切片，也可能是标量字符串，用 first-term 统一取首项 */}} {{- $term := partial "components/first-term" $ctx.Params.categories -}} {{- if not $term -}}{{- $term = partial "components/first-term" $ctx.Params.tags -}}{{- end -}} {{- with $term -}}{{- $key = . -}}{{- end -}} {{- $key = printf "%v" $key -}} {{- $day := 1 -}} {{- with $ctx.Date -}}{{- $day = int (.Format "2") -}}{{- end -}} {{- $seed := mod (add (len $key) $day) 5 -}} {{- $variant := mod (len $title) 3 -}} {{- $char := "·" -}} {{- if gt (len $title) 0 -}}{{- $char = slicestr $title 0 1 -}}{{- end -}} {{- $w := 320 -}} {{- $h := 180 -}} {{- if eq $ratio "1:1" -}}{{- $w = 240 -}}{{- $h = 240 -}}{{- end -}} {{- if eq $ratio "4:3" -}}{{- $w = 320 -}}{{- $h = 240 -}}{{- end -}} {{- if eq $ratio "21:9" -}}{{- $w = 420 -}}{{- $h = 180 -}}{{- end -}} {{/* 超大字位置：variant 0 右下 / variant 1 左下 / variant 2 居中偏右 */}} {{- $gx := sub $w 18 -}} {{- $gy := sub $h 30 -}} {{- $anchor := "end" -}} {{- if eq $variant 1 -}}{{- $gx = 22 -}}{{- $anchor = "start" -}} {{- else if eq $variant 2 -}}{{- $gx = add (div $w 2) 40 -}}{{- $anchor = "middle" -}}{{- end -}}
